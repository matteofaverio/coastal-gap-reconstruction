% ───────────────────────── SLIDE 11 ─────────────────────────
\begin{frame}[t]{Limitations}
  \vskip10pt
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      {\small\bfseries\color{accent}Validation and model limits}
      \vskip0.16cm
      {\footnotesize
      \begin{itemize}\setlength\itemsep{0.20cm}
        \item Real outages such as the 256-day gap exceed the validated gap lengths.
        \item Only one pretrained sequence architecture was tested.
        \item Event-day and tail-regime uncertainty calibration remain incomplete.
      \end{itemize}}
    \end{column}
    \begin{column}{0.49\textwidth}
      {\small\bfseries\color{accent}Physical and scope limits}
      \vskip0.16cm
      {\footnotesize
      \begin{itemize}\setlength\itemsep{0.20cm}
        \item External products are coarser than the buoy-scale signal.
        \item Covariate evidence is correlational, not causal.
        \item Validation is single-station; generality across CEAZA stations is untested.
      \end{itemize}}
    \end{column}
  \end{columns}
  \vskip0.30cm
  \takeaway{\centering{\bfseries\color{accent}Validate against strong baselines, preserve physical
    context, and examine where average skill fails.}}
\end{frame}

\begin{frame}[t]{Backup --- TS-ICL input and architecture}
  \vskip3pt
  \begin{columns}[T]
    \begin{column}{0.36\textwidth}
      \vskip0.15cm
      \centering
      \includegraphics[width=0.88\linewidth]{figures/tsicl_architecture_crop.pdf}
    \end{column}
    \begin{column}{0.60\textwidth}
      \vskip0.1cm
      {\footnotesize
      \begin{itemize}\setlength\itemsep{0.10cm}
        \item Visible target context $\mathbf y_{\mathrm{ctxt}}$ plus the masked hidden interval
          form one timestamp--value sequence; covariates $\mathbf X_{\mathrm{covar}}$ enter as
          time-indexed channels.
        \item Timestamp--value pairs and covariates are encoded, then \textbf{aggregated across
          channels} into timestamp-specific context representations.
        \item An \textbf{in-context regressor} (causal self-attention) reconstructs the hidden
          interval: $\hat{\mathbf y}_G=F(\mathbf y_{\mathrm{ctxt}},\mathbf X_{\mathrm{covar}})$.
        \item Output includes \textbf{quantile predictions} (q05--q95), not just a point estimate.
        \item Used \textbf{zero-shot}: pretrained elsewhere, no Tongoy-specific fitting.
      \end{itemize}}
    \end{column}
  \end{columns}
  \srcline{TS-ICL: Le Naour, Nabil \& Petralia (2026); software: EDF Lab (2026).}
\end{frame}
